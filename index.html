SQL и настройка

Обязательно
1
Создай бесплатный проектsupabase.com → New project. Дождись статуса Active.
2
Вставь ключи в игруSettings → API → Project URL и anon public → экран «Подключение Supabase».
3
Realtime должен быть включёнRealtime Settings → «Enable Realtime» = Enabled и «Allow public access to channels» = Enabled. Это значения по умолчанию. Мультиплеер работает уже на этом шаге: комнаты, очередь и бой идут через Broadcast + Presence.
Необязательно · реестр комнат в БД
SQL-блоки 1 и 2 включают реестр комнат в БД: приватные комнаты не читаются публичным списком, запись подтверждается токеном владельца. Блок 2 ограничивает публичный список 20 активными комнатами и возвращает понятную ошибку при заполнении. Без SQL игра использует Presence как временный список, но общий лимит 20 на уровне базы и постоянный реестр не действуют.
-- NEON ARENA · реестр комнат + безопасный доступ (SQL Editor → Run)
create extension if not exists pgcrypto;

create table if not exists public.arena_rooms (
  code        text primary key,
  name        text not null,
  host_id     text not null,
  host_nick   text not null,
  max_players int  not null default 4,
  players     int  not null default 1,
  is_private  boolean not null default false,
  in_game     boolean not null default false,
  token_hash  text not null,
  created_at  timestamptz not null default now(),
  updated_at  timestamptz not null default now()
);

alter table public.arena_rooms enable row level security;
alter table public.arena_rooms replica identity full;

-- клиент может только читать и только публичные комнаты
revoke insert, update, delete on public.arena_rooms from anon, authenticated;
grant  select on public.arena_rooms to anon, authenticated;

drop policy if exists arena_read_public on public.arena_rooms;
create policy arena_read_public on public.arena_rooms
  for select to anon, authenticated
  using (is_private = false);
-- Функции записи: владелец комнаты подтверждается секретным токеном
create or replace function public.arena_create_room(
  p_code text, p_name text, p_host_id text, p_host_nick text,
  p_max int, p_private boolean, p_token text)
returns boolean language plpgsql security definer
set search_path = public as $$
declare n int;
begin
  -- serialize public-room creation so two clients cannot exceed the 20-room cap
  perform pg_advisory_xact_lock(739184);
  delete from public.arena_rooms
   where updated_at < now() - interval '2 minutes';
  if not coalesce(p_private,false) then
    select count(*) into n from public.arena_rooms
     where is_private = false
       and updated_at > now() - interval '60 seconds';
    if n >= 20 then
      raise exception 'ROOM_LIMIT_20: maximum 20 active public rooms';
    end if;
  end if;
  insert into public.arena_rooms
    (code, name, host_id, host_nick, max_players, players,
     is_private, in_game, token_hash)
  values
    (upper(p_code), left(p_name,16), p_host_id, left(p_host_nick,14),
     greatest(2, least(4, coalesce(p_max,4))), 1,
     coalesce(p_private,false), false,
     encode(digest(p_token,'sha256'),'hex'))
  on conflict (code) do nothing;
  get diagnostics n = row_count;
  return n > 0;
end $$;

create or replace function public.arena_touch_room(
  p_code text, p_token text, p_players int, p_in_game boolean, p_host_nick text)
returns boolean language plpgsql security definer
set search_path = public as $$
declare n int;
begin
  update public.arena_rooms
     set players   = greatest(0, least(4, coalesce(p_players,1))),
         in_game   = coalesce(p_in_game,false),
         host_nick = left(coalesce(p_host_nick, host_nick),14),
         updated_at = now()
   where code = upper(p_code)
     and token_hash = encode(digest(p_token,'sha256'),'hex');
  get diagnostics n = row_count;
  return n > 0;
end $$;

create or replace function public.arena_close_room(p_code text, p_token text)
returns boolean language plpgsql security definer
set search_path = public as $$
declare n int;
begin
  delete from public.arena_rooms
   where code = upper(p_code)
     and token_hash = encode(digest(p_token,'sha256'),'hex');
  get diagnostics n = row_count;
  return n > 0;
end $$;

create or replace function public.arena_list()
returns table(code text, name text, host_nick text,
              max_players int, players int, in_game boolean)
language plpgsql security definer
set search_path = public as $$
begin
  delete from public.arena_rooms
   where updated_at < now() - interval '2 minutes';
  return query
    select r.code, r.name, r.host_nick, r.max_players, r.players, r.in_game
      from public.arena_rooms r
     where r.is_private = false
       and r.updated_at > now() - interval '60 seconds'
     order by r.updated_at desc
     limit 20;
end $$;

create or replace function public.arena_probe(p_code text)
returns table(ok boolean, priv boolean, players int, maxp int, nm text, ig boolean)
language plpgsql security definer
set search_path = public as $$
begin
  return query
    select true, r.is_private, r.players, r.max_players, r.name, r.in_game
      from public.arena_rooms r
     where r.code = upper(p_code)
       and r.updated_at > now() - interval '60 seconds';
  if not found then
    return query select false, false, 0, 0, ''::text, false;
  end if;
end $$;

revoke all on function public.arena_create_room(text,text,text,text,int,boolean,text) from public;
revoke all on function public.arena_touch_room(text,text,int,boolean,text) from public;
revoke all on function public.arena_close_room(text,text) from public;
revoke all on function public.arena_list() from public;
revoke all on function public.arena_probe(text) from public;

grant execute on function public.arena_create_room(text,text,text,text,int,boolean,text) to anon, authenticated;
grant execute on function public.arena_touch_room(text,text,int,boolean,text) to anon, authenticated;
grant execute on function public.arena_close_room(text,text) to anon, authenticated;
grant execute on function public.arena_list() to anon, authenticated;
grant execute on function public.arena_probe(text) to anon, authenticated;
